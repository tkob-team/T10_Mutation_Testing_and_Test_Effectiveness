  db.get(
    "SELECT * FROM coupons WHERE code = ? AND is_active = 1",
    [code],
    (err, coupon) => {
      if (!coupon) {
        return res
          .status(404)
          .json({ error: "Mã giảm giá không tồn tại hoặc đã bị vô hiệu hóa" });
      }

        const now = new Date();
